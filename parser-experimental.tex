\keys_define:nn { parserlpalign }
  {
    vars .tl_set:N = \lpalign_vars_text,
    vars-style .tl_set:N = \lpalign_vars_style,
    sense-shift .dim_set:N = \lpalign_sense_shift,
    st .tl_set:N = \lpalign_st_text,
    st-shift .dim_set:N = \lpalign_st_shift,
    objective-sep .dim_set:N = \lpalign_objective_sep,
    qualifier-sep .dim_set:N = \lpalign_qualifier_sep,
    gutter-sep .dim_set:N = \lpalign_gutter_sep,
  }

\cs_new_protected:Npn \lpalign_reset:
  {
    \tl_clear:N \lpalign_vars_text
    \tl_set:Nn \lpalign_vars_style { \scriptstyle }
    \tl_set:Nn \lpalign_st_text { s.t. }
    \dim_set:Nn \lpalign_sense_shift { 0pt }
    \dim_set:Nn \lpalign_st_shift { 0pt }
    \dim_set:Nn \lpalign_objective_sep { 0em }
    \dim_set:Nn \lpalign_qualifier_sep { 2em }
    \dim_set:Nn \lpalign_gutter_sep { 0.75em }
    \tl_clear:N \l_lpalign_objective_tl
  }

\cs_new_protected:Npn \lpalign_setup:nn #1#2
  {
    \lpalign_reset:
    \keys_set:nn { parserlpalign } { #1 }
    \hbox_set:Nn \lpalign_sense_box
      {
        \stackunder
          { #2 }
          {
            \tl_if_blank:VF \lpalign_vars_text
              { $\tl_use:N \lpalign_vars_style \tl_use:N \lpalign_vars_text$ }
          }
      }
  }

\cs_new:Npn \lpalign_print_sense:
  {
    \raisebox{\dim_use:N \lpalign_sense_shift}{\box_use:N \lpalign_sense_box}
  }

\cs_new:Npn \lpalign_print_st:
  {
    \raisebox{\dim_use:N \lpalign_st_shift}{\tl_use:N \lpalign_st_text}
  }

\cs_new:Npn \lpalign_parser:n #1
	{
	% End loop if token list is empty
	\tl_if_empty:nF { #1 }
		{
		% Trim leading spaces
		\tl_if_head_is_space:nTF { #1 }
			{
			\tl_gput_right:Nn \lp_parser_output { ~ }
			\exp_args:Ne \lpalign_parser:n { \tl_trim_spaces:n { #1 } }
			}
			{
			% Check if the head is a group
			\tl_if_head_is_group:nTF { #1 }
				{
				\tl_gput_right:Nx \lp_parser_output  { {\tl_head:n {#1}} }
				\exp_args:Ne \lpalign_parser:n { \tl_tail:n {#1} }
				}
				{
				\tl_gset:Nx \g_temp_head_tl { \tl_head:n {#1} }
				% Resolve nest depth
				\str_if_eq:eeT { \tl_to_str:n { \begin } } { \tl_to_str:V \g_temp_head_tl }
					{ \int_gincr:N \g_nest_depth }
				\str_if_eq:eeT { \tl_to_str:n { \end } } { \tl_to_str:V \g_temp_head_tl }
					{ \int_gdecr:N \g_nest_depth }
        % Replace depth zero double ampersands with qualifier spacing
        \str_if_eq:eeTF { \tl_to_str:n { & } } { \tl_to_str:V \g_temp_head_tl }
          {
          \str_if_eq:eeTF { \tl_to_str:n { & } } { \exp_args:Ne \tl_to_str:n { \tl_head:n { \tl_tail:n {#1} } } }
            {
            \int_compare:nTF { \g_nest_depth > 0 }
              { \tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} } }
              {
              \tl_gput_right:Nn \lp_parser_output { & }
              \tl_gput_right:Nx \lp_parser_output { \exp_not:N \hspace { \dim_use:N \lpalign_qualifier_sep } }
              \tl_gput_right:Nn \lp_parser_output { & }
              \exp_args:Ne \lpalign_parser:n { \tl_tail:n { \tl_tail:n {#1} } }
              }
            }
            {
            % Check if token is a line break
            \str_if_eq:eeTF { \tl_to_str:n { \\ } } { \tl_to_str:V \g_temp_head_tl }
              {
              % Replace depth zero linebreaks
              \int_compare:nTF { \g_nest_depth > 0 }
                {\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
                {\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} &&}}
              }
              {\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
            \exp_args:Ne \lpalign_parser:n { \tl_tail:n {#1} }
            }
          }
          {
				% Check if token is a line break
				\str_if_eq:eeTF { \tl_to_str:n { \\ } } { \tl_to_str:V \g_temp_head_tl }
					{
					% Replace depth zero linebreaks
					\int_compare:nTF { \g_nest_depth > 0 }
						{\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
						{\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} &&}}
					}
					{\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
				\exp_args:Ne \lpalign_parser:n { \tl_tail:n {#1} }
				}
          }
			}
		}
	}

	{
	\lpalign_setup:nn { #1 } { #2 }
  \tl_set:Nn \l_lpalign_objective_tl { #3 }
	\tl_gclear:N \lp_parser_output
	\int_gzero:N \g_nest_depth
	\lpalign_parser:n { #4 }
	\begin{alignat}{3}
	\lpalign_print_sense:
  &\hspace{\dim_use:N \lpalign_gutter_sep}&
  \tl_use:N \l_lpalign_objective_tl
  \\[\dim_use:N \lpalign_objective_sep]
	\lpalign_print_st: && 
	\tl_use:N \lp_parser_output
	\end{alignat}
	}
	{\ignorespacesafterend}
