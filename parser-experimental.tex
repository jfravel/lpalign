\ExplSyntaxOn


\tl_new:N \lp_parser_output
\tl_new:N \g_temp_head_tl
\int_new:N \g_nest_depth
\tl_new:N \l_lpalign_vars_tl
\tl_new:N \l_lpalign_st_tl
\dim_new:N \l_lpalign_qualifier_sep_dim

\keys_define:nn { parserlpalign }
  {
    vars .tl_set:N = \l_lpalign_vars_tl,
    st .tl_set:N = \l_lpalign_st_tl,
    qualifier-sep .dim_set:N = \l_lpalign_qualifier_sep_dim,
  }

\cs_new_protected:Npn \lpalign_reset:
  {
    \tl_clear:N \l_lpalign_vars_tl
    \tl_set:Nn \l_lpalign_st_tl { subject\ to }
    \dim_set:Nn \l_lpalign_qualifier_sep_dim { 2em }
  }

\cs_new:Npn \lpalign_parser:n #1
	{
	% End loop if token list is empty
	\tl_if_empty:nF { #1 }
		{
		% Trim leading spaces
		\tl_if_head_is_space:nTF { #1 }
			{
			\tl_gput_right:Nn \lp_parser_output { ~ }
			\exp_args:Ne \lpalign_parser:n { \tl_trim_spaces:n { #1 } }
			}
			{
			% Check if the head is a group
			\tl_if_head_is_group:nTF { #1 }
				{
				\tl_gput_right:Nx \lp_parser_output  { {\tl_head:n {#1}} }
				\exp_args:Ne \lpalign_parser:n { \tl_tail:n {#1} }
				}
				{
				\tl_gset:Nx \g_temp_head_tl { \tl_head:n {#1} }
				% Resolve nest depth
				\str_if_eq:eeT { \tl_to_str:n { \begin } } { \tl_to_str:V \g_temp_head_tl }
					{ \int_gincr:N \g_nest_depth }
				\str_if_eq:eeT { \tl_to_str:n { \end } } { \tl_to_str:V \g_temp_head_tl }
					{ \int_gdecr:N \g_nest_depth }
				% Check if token is a line break
				\str_if_eq:eeTF { \tl_to_str:n { \\ } } { \tl_to_str:V \g_temp_head_tl }
					{
					% Replace depth zero linebreaks
					\int_compare:nTF { \g_nest_depth > 0 }
						{\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
						{
						\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }
						\tl_gput_right:Nn \lp_parser_output { & }
						\tl_gput_right:Nx \lp_parser_output { \exp_not:N \hspace { \dim_use:N \l_lpalign_qualifier_sep_dim } }
						\tl_gput_right:Nn \lp_parser_output { & }
						}
					}
					{\tl_gput_right:Nx \lp_parser_output { \tl_head:n {#1} }}
				\exp_args:Ne \lpalign_parser:n { \tl_tail:n {#1} }
				}
			}
		}
	}



\NewDocumentCommand{\parseDisplay}{m}
	{
	 \tl_gclear:N \lp_parser_output
	 \lpalign_parser:n { #1 }
	 % Output the compiled token list to the page
%	 \tl_log:N \lp_parser_output
	 $$\tl_use:N \lp_parser_output$$
	}
	
\NewDocumentEnvironment{lpalign}{O{}mm+b}
	{
	\lpalign_reset:
	\keys_set:nn { parserlpalign } { #1 }
	\tl_gclear:N \lp_parser_output
	\int_gzero:N \g_nest_depth
	\lpalign_parser:n { #4 }
	\begin{alignat}{3}
	\mathrm{#2}
  \tl_if_blank:VF \l_lpalign_vars_tl
    { _{\l_lpalign_vars_tl} }
  &\quad&
  #3
  \\
	\mathrm{\tl_use:N \l_lpalign_st_tl} && 
	\tl_use:N \lp_parser_output
	\end{alignat}
	}
	{\ignorespacesafterend}
