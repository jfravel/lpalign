\documentclass{article}
\usepackage{lpalign}
\usepackage{hyperref}

\begin{document}

\section*{Default}
\begin{namedsubeqs}{EQ}
\begin{align}
  A\mathbf{x} &= \mathbf{b} \label{eq:1}\\
  C\mathbf{y} &= \mathbf{d} \label{eq:2}
\end{align}
\end{namedsubeqs}

\noindent
\verb|\ref{eq:1}|: \ref{eq:1}\\
\verb|\subref{eq:1}|: \subref{eq:1}\\
\verb|\subeqref{eq:2}|: \subeqref{eq:2}

\section*{Legacy Shorthand}
\begin{namedsubeqs}{LEG}[\roman]
\begin{align}
  E\mathbf{x} &= \mathbf{f} \label{leg:1}\\
  G\mathbf{y} &= \mathbf{h} \label{leg:2}
\end{align}
\end{namedsubeqs}

\noindent
\verb|\ref{leg:1}|: \ref{leg:1}\\
\verb|\subref{leg:1}|: \subref{leg:1}\\
\verb|\subeqref{leg:2}|: \subeqref{leg:2}

\section*{Key-Value Style}
\begin{namedsubeqs}{KV}[style=\itshape\roman]
\begin{align}
  I\mathbf{x} &= \mathbf{j} \label{kv:1}\\
  K\mathbf{y} &= \mathbf{l} \label{kv:2}
\end{align}
\end{namedsubeqs}

\noindent
\verb|\ref{kv:1}|: \ref{kv:1}\\
\verb|\subref{kv:1}|: \subref{kv:1}\\
\verb|\subeqref{kv:2}|: \subeqref{kv:2} Text after reference.

\section*{Key-Value Punctuation}
\begin{namedsubeqs}{NLP}[style=\roman,punct=-]
\begin{align}
  M\mathbf{x} &= \mathbf{n} \label{nlp:1}\\
  O\mathbf{y} &= \mathbf{p} \label{nlp:2}
\end{align}
\end{namedsubeqs}

\noindent
\verb|\ref{nlp:1}|: \ref{nlp:1}\\
\verb|\subref{nlp:1}|: \subref{nlp:1}\\
\verb|\subeqref{nlp:2}|: \subeqref{nlp:2}
