\section*{Default Alpha}
\begin{subequations}\label{alpha}
\begin{align}
  A\mathbf{x} &= \mathbf{b} \label{alpha:1}\\
  C\mathbf{y} &= \mathbf{d} \label{alpha:2}
\end{align}
\end{subequations}

\noindent
\verb|\ref{alpha}|: \ref{alpha}\\
\verb|\ref{alpha:1}|: \ref{alpha:1}\\
\verb|\subref{alpha:1}|: \subref{alpha:1}\\
\verb|\subeqref{alpha:2}|: \subeqref{alpha:2}
