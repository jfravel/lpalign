\begin{lpalign}[
  vars={x \in \mathbb{R}^n},
  sense-pos=c,
  st-pos=t,
  st-shift=-0.5\baselineskip
]{Maximize}{
  \begin{aligned}
    c_1x_1 + c_2x_2 + c_3x_3 \\
    {} + c_4x_4 + \cdots + c_nx_n
  \end{aligned}
}
  \begin{bmatrix}
    a_{11} & \cdots & a_{1n} \\
    \vdots & \ddots & \vdots \\
    a_{m1} & \cdots & a_{mn}
  \end{bmatrix}
  \begin{bmatrix}
    x_1 \\ \vdots \\ x_n
  \end{bmatrix}
  &\le
  \begin{bmatrix}
    b_1 \\ \vdots \\ b_m
  \end{bmatrix} \\
  x &\ge 0
\end{lpalign}
